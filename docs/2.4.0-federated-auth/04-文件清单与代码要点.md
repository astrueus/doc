## F0 · Repo

新建 `internal/repository/member_identity_repo.go`：

```go
package repository

import (
	"context"
	"time"

	"git.itopcms.com/astrueus/doc/internal/model"
	"github.com/beego/beego/v2/client/orm"
)

type MemberIdentityRepo interface {
	FindByProviderSubject(ctx context.Context, provider, subject string) (*model.MemberIdentity, error)
	Insert(ctx context.Context, row *model.MemberIdentity) error
	TouchLastLogin(ctx context.Context, id int) error
}

type memberIdentityRepo struct{ defaultOrm orm.Ormer }

func NewMemberIdentityRepo(o orm.Ormer) MemberIdentityRepo {
	return &memberIdentityRepo{defaultOrm: o}
}

func (r *memberIdentityRepo) ormer(ctx context.Context) orm.Ormer {
	if r.defaultOrm != nil {
		return r.defaultOrm
	}
	return OrmFromContext(ctx)
}

func (r *memberIdentityRepo) FindByProviderSubject(ctx context.Context, provider, subject string) (*model.MemberIdentity, error) {
	row := model.NewMemberIdentity()
	err := r.ormer(ctx).QueryTable(row.TableNameWithPrefix()).
		Filter("provider", provider).Filter("subject", subject).One(row)
	if err != nil {
		return nil, err
	}
	return row, nil
}

func (r *memberIdentityRepo) Insert(ctx context.Context, row *model.MemberIdentity) error {
	id, err := r.ormer(ctx).Insert(row)
	if err != nil {
		return err
	}
	row.ID = int(id)
	return nil
}

func (r *memberIdentityRepo) TouchLastLogin(ctx context.Context, id int) error {
	_, err := r.ormer(ctx).QueryTable(model.NewMemberIdentity().TableNameWithPrefix()).
		Filter("id", id).Update(orm.Params{"last_login_at": time.Now()})
	return err
}
```

联邦建 `Member`：在 `MemberRepo` **新增** `CreateFederated`（不要调用 `Member.Add`）：

```go
// CreateFederated 插入联邦账号：字母数字 account、占位或真实 email、不可用密码哈希、auth_method=federated。
CreateFederated(ctx context.Context, m *model.Member) error
```

实现：校验 `RegexpAccount`；邮箱非空；`PasswordHash` 随机 32 字节；`AuthMethod = config.AuthMethodFederated`；`Status=0`；`Role` 来自入参；`o.Insert`。唯一冲突返回可包装错误。Binder 用事务：先 Insert Member，再 Insert identity（`orm.NewOrm().Begin` / `Commit`，Repo 方法接受同一 `ormer` 或 `BeginTx`——与现有 tx 单测同风格，见 `internal/repository` 已有 tx 测试）。

```go
type Binder struct {
	Members    repository.MemberRepo
	Identities repository.MemberIdentityRepo
	AutoCreate bool
	DefaultRole config.SystemRole
}

func (b *Binder) MemberFromIdentity(ctx context.Context, id *Identity) (*model.Member, error) {
	row, err := b.Identities.FindByProviderSubject(ctx, id.Provider, id.Subject)
	if err == nil {
		m, err := b.Members.Find(ctx, row.MemberID)
		if err != nil {
			return nil, err
		}
		if m.Status != 0 {
			return nil, errors.New("账号已被禁用")
		}
		_ = b.Identities.TouchLastLogin(ctx, row.ID)
		return m, nil
	}
	if !errors.Is(err, orm.ErrNoRows) {
		return nil, err
	}
	if !b.AutoCreate {
		return nil, errors.New("未绑定本地账号")
	}
	// 事务内 CreateFederated + Insert identity
	return m, nil
}
```

## F0 · Controller 与路由

新建 `internal/controller/FederatedLoginController.go`：

```go
type FederatedLoginController struct {
	BaseController
}

func (c *FederatedLoginController) Start() {
	c.Prepare()
	if !config.MustGlobal().Auth.FederatedEnable {
		c.ShowErrorPage(404, "未启用联邦登录")
		return
	}
	name := c.Ctx.Input.Param(":provider")
	p, ok := federated.Lookup(name)
	if !ok || !p.Enabled() {
		c.ShowErrorPage(404, "未知登录方式")
		return
	}
	state, err := federated.NewState()
	if err != nil {
		c.JsonResult(500, "生成 state 失败")
		return
	}
	c.SetSession("federated_state", state)
	c.SetSession("federated_provider", name)
	res, err := p.Start(c.Ctx.Request.Context(), state)
	if err != nil {
		c.JsonResult(500, err.Error())
		return
	}
	if res.Kind == "qr" {
		c.JsonResult(0, "ok", res)
		return
	}
	c.Redirect(res.URL, 302)
}

func (c *FederatedLoginController) Callback() {
	c.Prepare()
	name := c.Ctx.Input.Param(":provider")
	expect, _ := c.GetSession("federated_state").(string)
	expectP, _ := c.GetSession("federated_provider").(string)
	c.DelSession("federated_state")
	c.DelSession("federated_provider")
	got := c.GetString("state")
	if err := federated.ConsumeState(got, expect, name, expectP); err != nil {
		c.ShowErrorPage(403, "state 无效")
		return
	}
	p, ok := federated.Lookup(name)
	if !ok {
		c.ShowErrorPage(404, "未知登录方式")
		return
	}
	id, err := p.Complete(c.Ctx.Request.Context(), federated.Callback{
		Query: c.Ctx.Request.URL.Query(),
		Form:  c.Ctx.Request.Form,
	})
	if err != nil {
		c.ShowErrorPage(401, "换取身份失败")
		return
	}
	member, err := federatedBinder().MemberFromIdentity(c.Ctx.Request.Context(), id)
	if err != nil {
		c.ShowErrorPage(403, err.Error())
		return
	}
	c.SetMember(*member)
	// LoggedIn 在 AccountController 上；把 referer() 抽到 BaseController，或此处：
	turl, _ := url.PathUnescape(c.GetString("url"))
	if turl == "" {
		turl = config.URLFor("HomeController.Index")
	}
	c.Redirect(turl, 302)
}
```
